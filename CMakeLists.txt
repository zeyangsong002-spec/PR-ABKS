cmake_minimum_required(VERSION 3.18)
project(prabks_corrected LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(PBC_INCLUDE_DIR pbc.h PATH_SUFFIXES pbc REQUIRED)
find_library(PBC_LIBRARY pbc REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(CRYPTO_LIBRARY NAMES crypto libcrypto.so.3 REQUIRED)
add_executable(prabks src/main.cpp)
target_include_directories(prabks PRIVATE ${PBC_INCLUDE_DIR})
target_link_libraries(prabks PRIVATE ${PBC_LIBRARY} ${GMP_LIBRARY} ${CRYPTO_LIBRARY})
target_compile_options(prabks PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(prabks PRIVATE DEFAULT_PARAM_FILE="${CMAKE_CURRENT_SOURCE_DIR}/params/a.param")
option(ENABLE_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
if(ENABLE_SANITIZERS)
  target_compile_options(prabks PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(prabks PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME correctness COMMAND prabks --self-test)
